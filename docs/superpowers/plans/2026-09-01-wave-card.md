# Wave Card Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a reusable `uni-wave-card` component that faithfully recreates the supplied reference and integrate it as uni-card's third template across selection, editing, drafts, preview reuse, 2× export, archiving, and thumbnail display.

**Architecture:** Use one raster background asset derived from the supplied reference, then render the editable white frame and serif text natively in the component and with matching coordinates in Canvas. Keep the five-field template registry intact, move defaults and editor-row metadata into `config.js`, and isolate mini-program static render branches in one `card-renderer` host because Vue dynamic components are unavailable on MP-WEIXIN.

**Tech Stack:** uni-app, Vue 3 SFC, MP-WEIXIN Canvas 2D, Vitest, Image Gen, FFmpeg, Codex in-app Browser.

```js
const W = {
  w: 480,
  minH: 896,
  bg: '#f66b61',
  pageBg: '#f66b61',
  frameX: 47,
  frameY: 334,
  frameW: 386,
  frameMinH: 224,
  frameBottom: 338,
  framePadX: 22,
  framePadY: 30,
  frameRadius: 18,
  frameLine: 4,
  titleSize: 42,
  titleLH: 50,
  bodySize: 29,
  bodyLH: 34,
  titleBodyGap: 2,
}

function waveTextLines(ctx, text, maxWidth) {
  return String(text || '')
    .split('\n')
    .map((line) => line.trim())
    .filter(Boolean)
    .flatMap((line) => wrapRich(ctx, line, maxWidth))
}

export function measureWaveLayout(ctx, fields) {
  const textWidth = W.frameW - W.framePadX * 2
  setFont(ctx, 700, W.titleSize, 'Georgia, "Times New Roman", "Songti SC", serif')
  const titleLines = waveTextLines(ctx, fields.title, textWidth)
  setFont(ctx, 400, W.bodySize, 'Georgia, "Times New Roman", "Songti SC", serif')
  const contentLines = waveTextLines(ctx, fields.content, textWidth)
  const blockHeight =
    titleLines.length * W.titleLH +
    (titleLines.length && contentLines.length ? W.titleBodyGap : 0) +
    contentLines.length * W.bodyLH
  const frameHeight = Math.max(W.frameMinH, blockHeight + W.framePadY * 2)
  const height = Math.max(W.minH, W.frameY + frameHeight + W.frameBottom)
  return {
    width: W.w,
    height,
    frameX: W.frameX,
    frameY: W.frameY,
    frameWidth: W.frameW,
    frameHeight,
    blockHeight,
    titleLines,
    contentLines,
  }
}
```

- [ ] **Step 4: Add `paintWave`**

Add this function after `measureWaveLayout`:

```js
function paintWave(ctx, fields, imgs, draw) {
  const layout = measureWaveLayout(ctx, fields)
  if (!draw) return layout.height

  ctx.fillStyle = W.bg
  ctx.fillRect(0, 0, W.w, layout.height)
  if (imgs.waveBg) {
    ctx.drawImage(imgs.waveBg, 0, 0, W.w, layout.height)
  }

  ctx.save()
  ctx.strokeStyle = 'rgba(255, 250, 240, 0.96)'
  ctx.lineWidth = W.frameLine
  roundRect(
    ctx,
    layout.frameX,
    layout.frameY,
    layout.frameWidth,
    layout.frameHeight,
    W.frameRadius
  )
  ctx.stroke()
  ctx.restore()

  const blockTop = layout.frameY + (layout.frameHeight - layout.blockHeight) / 2
  ctx.textAlign = 'center'
  ctx.textBaseline = 'alphabetic'
  ctx.fillStyle = '#fffaf0'

  setFont(ctx, 700, W.titleSize, 'Georgia, "Times New Roman", "Songti SC", serif')
  let y = blockTop
  for (const line of layout.titleLines) {
    ctx.fillText(line, W.w / 2, y + W.titleSize)
    y += W.titleLH
  }

  if (layout.titleLines.length && layout.contentLines.length) y += W.titleBodyGap
  setFont(ctx, 400, W.bodySize, 'Georgia, "Times New Roman", "Songti SC", serif')
  for (const line of layout.contentLines) {
    ctx.fillText(line, W.w / 2, y + W.bodySize)
    y += W.bodyLH
  }
  ctx.textAlign = 'left'
  return layout.height
}
```
